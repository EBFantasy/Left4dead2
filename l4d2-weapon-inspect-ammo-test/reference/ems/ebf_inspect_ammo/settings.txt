// output_chat     0 or 1. Ammo line in the chat area. Default 1.
// output_center   0 or 1. Ammo line at screen center. Default 1.
// play_animation  0 or 1. Drive the weapon's reload/inspect animation.
//                 Custom weapon models that ship an inspect animation
//                 will show it. Stock models show their reload. Default 1.
// block_reload    0 or 1. While E is held, report the magazine as full
//                 so the engine refuses to reload. Default 1.
//                 The true ammo count is restored when E is released,
//                 and is what the readout always shows.
//                 Requires require_use 1.
// guard_ticks     1 to 40. Backup only: frames an ammo snapshot is
//                 restored if a reload slips through. Default 8.
// cancel_window   0.5 to 10.0. Seconds the fallback keeps cancelling
//                 a reload, covering the whole reload animation.
//                 Default 3.0. Raise for very slow custom reloads.
// cooldown        0.0 to 10.0. Seconds between inspects. Default 1.20.
// melee_ok        0 or 1. Allow inspecting melee and clipless items.
//                 Default 1.
// debug           0 or 1. Verbose console diagnostics. Default 0.
// ============================================================

output_chat 1
output_center 1
play_animation 1
block_reload 1
guard_ticks 8
cancel_window 3.0
